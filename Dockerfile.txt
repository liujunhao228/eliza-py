# Eliza-Py Dockerfile
# 支持 Alice 聊天机器人和 Turing 测试后端
# 使用多阶段构建优化镜像大小和构建速度
# 包含前端构建，单容器部署

# =============================================================================
# 阶段 1: 前端构建 - 构建 Vue 前端
# =============================================================================
FROM node:20-alpine AS frontend-builder

WORKDIR /app/turing_test/frontend

# 复制 package files 利用层缓存
COPY turing_test/frontend/package*.json ./

# 安装依赖（使用 npm ci 确保可重复构建）
RUN npm ci --silent

# 复制源代码
COPY turing_test/frontend/ ./

# 创建生产环境配置
RUN echo "VITE_API_BASE_URL=/api" > .env.production && \
    echo "VITE_WS_BASE_URL=ws://localhost/ws" >> .env.production

# 构建前端
RUN npm run build

# =============================================================================
# 阶段 2: Python 依赖构建
# =============================================================================
FROM python:3.12.12-slim AS builder

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_NO_CACHE_DIR=1 \
    PIP_DISABLE_PIP_VERSION_CHECK=1

WORKDIR /app

# 安装构建依赖和 uv（使用单行减少层数）
RUN apt-get update && apt-get install -y --no-install-recommends \
    gcc python3-dev \
    && rm -rf /var/lib/apt/lists/* \
    && apt-get clean

# 安装 uv
RUN pip install --no-cache-dir uv

# 仅复制依赖文件（利用层缓存）
COPY pyproject.toml uv.lock ./

# 创建虚拟环境并安装依赖
RUN uv venv /opt/venv && \
    UV_PROJECT_ENVIRONMENT=/opt/venv uv sync --frozen --no-dev

# 复制 uv 到标准路径
RUN cp /usr/local/bin/uv /opt/uv

ENV VIRTUAL_ENV=/opt/venv \
    PATH="/opt/venv/bin:$PATH"

# =============================================================================
# 阶段 3: 运行阶段（最小化镜像）
# =============================================================================
FROM python:3.12.12-slim AS runner

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_NO_CACHE_DIR=1 \
    PIP_DISABLE_PIP_VERSION_CHECK=1

WORKDIR /app

# 单行安装运行时依赖（减少层数）
RUN apt-get update && apt-get install -y --no-install-recommends \
    nginx libgomp1 bash \
    && rm -rf /var/lib/apt/lists/* \
    && apt-get clean \
    && rm -rf /var/cache/apt/archives/*

# 从构建阶段复制虚拟环境和 uv
COPY --from=builder /opt/venv /opt/venv
COPY --from=builder /opt/uv /usr/local/bin/uv
ENV VIRTUAL_ENV=/opt/venv \
    PATH="/opt/venv/bin:$PATH"

# 从前端构建阶段复制静态文件
COPY --from=frontend-builder /app/turing_test/backend/static /usr/share/nginx/html

# 复制 nginx 配置
COPY docker/nginx.conf /etc/nginx/nginx.conf

# 复制应用代码（放在最后以利用层缓存）
COPY . .

# 复制入口脚本
COPY docker/entrypoint.sh /usr/local/bin/entrypoint
RUN chmod +x /usr/local/bin/entrypoint

# 创建非 root 用户和组
RUN groupadd --gid 1001 appgroup && \
    useradd --uid 1001 --gid appgroup --shell /bin/bash --create-home appuser

# 创建目录并设置权限（在 COPY . . 之后）
RUN mkdir -p /app/data /app/logs /app/scripts/lua && \
    chown -R appuser:appgroup /app /opt/venv /usr/share/nginx/html && \
    chmod 755 /app/data && \
    mkdir -p /var/log/nginx /var/cache/nginx /run /tmp/nginx_client_body /tmp/nginx_proxy /tmp/nginx_fastcgi /tmp/nginx_uwsgi /tmp/nginx_scgi && \
    chown -R appuser:appgroup /var/log/nginx /var/cache/nginx /run /tmp/nginx_client_body /tmp/nginx_proxy /tmp/nginx_fastcgi /tmp/nginx_uwsgi /tmp/nginx_scgi

# 暴露端口
EXPOSE 3001 8080

# 健康检查
HEALTHCHECK --interval=30s --timeout=10s --start-period=10s --retries=3 \
    CMD python -c "import socket; s=socket.socket(); s.settimeout(5); result=s.connect_ex(('127.0.0.1', 3001)); s.close(); exit(0 if result==0 else 1)" || exit 1

# 入口点和默认命令
ENTRYPOINT ["/usr/local/bin/entrypoint"]
CMD ["python", "main.py", "turing"]

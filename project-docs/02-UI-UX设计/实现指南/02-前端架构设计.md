## 一、单页应用（SPA）架构

### 1.1 架构概述

采用经典的单页应用（SPA）架构，所有交互都在客户端完成，通过AJAX和WebSocket与后端通信。

```
┌─────────────────────────────────────────────────┐
│                  用户界面 (UI)                  │
│  ┌───────────┐ ┌───────────┐ ┌───────────┐     │
│  │  组件A    │ │  组件B    │ │  组件C    │     │
│  └───────────┘ └───────────┘ └───────────┘     │
└─────────────────────────────────────────────────┘
                       ↓
┌─────────────────────────────────────────────────┐
│              状态管理层 (Pinia Stores)           │
│  ┌───────────┐ ┌───────────┐ ┌───────────┐     │
│  │ User Store│ │Game Store │ │Session... │     │
│  └───────────┘ └───────────┘ └───────────┘     │
└─────────────────────────────────────────────────┘
                       ↓
┌─────────────────────────────────────────────────┐
│              业务逻辑层 (Composables)           │
│  ┌───────────┐ ┌───────────┐ ┌───────────┐     │
│  │ useWebSock│ │useMetaConv│ │useScore...│     │
│  └───────────┘ └───────────┘ └───────────┘     │
└─────────────────────────────────────────────────┘
                       ↓
┌─────────────────────────────────────────────────┐
│               数据访问层 (API)                   │
│  ┌───────────┐ ┌───────────┐ ┌───────────┐     │
│  │  axios    │ │ WebSocket │ │  utils    │     │
│  └───────────┘ └───────────┘ └───────────┘     │
└─────────────────────────────────────────────────┘
                       ↓
┌─────────────────────────────────────────────────┐
│                 后端 API / WebSocket            │
└─────────────────────────────────────────────────┘
```

### 1.2 数据流向

#### 单向数据流（推荐）

```
用户操作 → 组件事件 → Action → 状态更新 → 组件重渲染
```

#### 双向数据绑定（表单等）

```
表单输入 ↔ v-model ↔ 响应式数据
```

#### 3. 事件总线（Event Bus）

```typescript
// utils/eventBus.ts
import { ref } from 'vue'

const eventBus = ref<Map<string, Function[]>>(new Map())

export function useEventBus() {
  function on(event: string, callback: Function) {
    if (!eventBus.value.has(event)) {
      eventBus.value.set(event, [])
    }
    eventBus.value.get(event)!.push(callback)
  }
  
  function emit(event: string, data?: any) {
    const callbacks = eventBus.value.get(event) || []
    callbacks.forEach(callback => callback(data))
  }
  
  function off(event: string, callback: Function) {
    const callbacks = eventBus.value.get(event) || []
    const index = callbacks.indexOf(callback)
    if (index > -1) {
      callbacks.splice(index, 1)
    }
  }
  
  return { on, emit, off }
}
```

### 2.3 组件设计原则

#### 1. 单一职责原则
每个组件只负责一个明确的功能。

#### 2. 组件粒度
- **原子组件**：按钮、输入框等
- **分子组件**：表单项、卡片等
- **组织组件**：页面级组件

#### 3. 可复用性
通过props和slots提高组件复用性。

```vue
<!-- 可复用的按钮组件 -->
<template>
  <button 
    :class="['btn', `btn-${type}`, { 'btn-loading': loading }]"
    :disabled="disabled || loading"
    @click="handleClick"
  >
    <span v-if="loading" class="spinner"></span>
    <slot />
  </button>
</template>

<script setup lang="ts">
defineProps<{
  type?: 'primary' | 'secondary' | 'accent' | 'danger'
  disabled?: boolean
  loading?: boolean
}>()

const emit = defineEmits<{
  (e: 'click', event: MouseEvent): void
}>()

function handleClick(event: MouseEvent) {
  emit('click', event)
}
</script>
```

#### 4. 性能优化
- 使用`v-once`静态内容
- 使用`v-memo`避免不必要的重渲染
- 虚拟滚动长列表

### 3.1 Store设计

#### User Store - 用户状态

```typescript
// stores/user.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const useUserStore = defineStore('user', () => {
  // 状态
  const userId = ref<string>('')
  const nickname = ref<string>('')
  const score = ref<number>(100)
  const isLoggedIn = ref<boolean>(false)
  const inviteCode = ref<string>('')
  
  // 统计数据
  const totalSessions = ref<number>(0)
  const highestScore = ref<number>(100)
  const lowestScore = ref<number>(100)
  const avgTurns = ref<number>(0)
  const avgConfidence = ref<string>('mid')
  
  // 计算属性
  const canStartNewSession = computed(() => {
    return score.value >= 2 // 入场券成本
  })
  
  // Actions
  function login(data: { userId: string; nickname: string; inviteCode: string }) {
    userId.value = data.userId
    nickname.value = data.nickname
    inviteCode.value = data.inviteCode
    isLoggedIn.value = true
  }
  
  function updateScore(delta: number) {
    score.value += delta
    if (score.value > highestScore.value) {
      highestScore.value = score.value
    }
    if (score.value < lowestScore.value) {
      lowestScore.value = score.value
    }
  }
  
  function incrementSessionCount() {
    totalSessions.value++
  }
  
  function logout() {
    userId.value = ''
    nickname.value = ''
    score.value = 100
    isLoggedIn.value = false
    inviteCode.value = ''
  }
  
  return {
    // 状态
    userId,
    nickname,
    score,
    isLoggedIn,
    inviteCode,
    totalSessions,
    highestScore,
    lowestScore,
    avgTurns,
    avgConfidence,
    
    // 计算属性
    canStartNewSession,
    
    // Actions
    login,
    updateScore,
    incrementSessionCount,
    logout
  }
})
```

#### Game Store - 博弈状态

```typescript
// stores/game.ts
import { defineStore } from 'pinia'
import { ref, reactive } from 'vue'

interface Message {
  id: string
  sender: 'user' | 'opponent' | 'system'
  content: string
  timestamp: Date
  isMetaConversation: boolean
}

interface ScorePrediction {
  lowConfidence: { correct: number; wrong: number }
  midConfidence: { correct: number; wrong: number }
  highConfidence: { correct: number; wrong: number }
  metaMultiplier: number
  penaltyMultiplier: number
}

export const useGameStore = defineStore('game', () => {
  // 状态
  const turn = ref<number>(0)
  const metaConversationCount = ref<number>(0)
  const messages = ref<Message[]>([])
  const triggeredMidGame = ref<boolean>(false)
  const entryFee = ref<number>(2)
  const turnPenalty = ref<number>(0)
  
  const currentScorePrediction = reactive<ScorePrediction>({
    lowConfidence: { correct: 0, wrong: 0 },
    midConfidence: { correct: 0, wrong: 0 },
    highConfidence: { correct: 0, wrong: 0 },
    metaMultiplier: 1.0,
    penaltyMultiplier: 1.0
  })
  
  // Actions
  function addMessage(message: Message) {
    messages.value.push(message)
  }
  
  function incrementTurn() {
    turn.value++
    updateTurnPenalty()
    updateScorePrediction()
  }
  
  function incrementMetaCount() {
    metaConversationCount.value++
    updateScorePrediction()
  }
  
  function updateTurnPenalty() {
    const MIN_FREE_TURNS = 3
    const TURN_PENALTY_RATE = 0.5
    turnPenalty.value = Math.max(0, (turn.value - MIN_FREE_TURNS) * TURN_PENALTY_RATE)
  }
  
  function updateScorePrediction() {
    const ENTRY_FEE = 2
    const metaMultiplier = 1 + (metaConversationCount.value * 0.2)
    const penaltyMultiplier = 1 + (metaConversationCount.value * 0.3)
    
    const baseReward = 10  // 识别AI正确
    const basePenalty = -15  // 误判AI为真人
    
    currentScorePrediction.lowConfidence = {
      correct: (baseReward * 1.0 * metaMultiplier) - ENTRY_FEE - turnPenalty.value,
      wrong: (basePenalty * 1.0 * penaltyMultiplier) - ENTRY_FEE - turnPenalty.value
    }
    
    currentScorePrediction.midConfidence = {
      correct: (baseReward * 2.5 * metaMultiplier) - ENTRY_FEE - turnPenalty.value,
      wrong: (basePenalty * 2.5 * penaltyMultiplier) - ENTRY_FEE - turnPenalty.value
    }
    
    currentScorePrediction.highConfidence = {
      correct: (baseReward * 5.0 * metaMultiplier) - ENTRY_FEE - turnPenalty.value,
      wrong: (basePenalty * 5.0 * penaltyMultiplier) - ENTRY_FEE - turnPenalty.value
    }
    
    currentScorePrediction.metaMultiplier = metaMultiplier
    currentScorePrediction.penaltyMultiplier = penaltyMultiplier
  }
  
  function triggerMidGameJudgment() {
    triggeredMidGame.value = true
  }
  
  function reset() {
    turn.value = 0
    metaConversationCount.value = 0
    messages.value = []
    triggeredMidGame.value = false
    turnPenalty.value = 0
    updateScorePrediction()
  }
  
  return {
    // 状态
    turn,
    metaConversationCount,
    messages,
    triggeredMidGame,
    entryFee,
    turnPenalty,
    currentScorePrediction,
    
    // Actions
    addMessage,
    incrementTurn,
    incrementMetaCount,
    updateTurnPenalty,
    updateScorePrediction,
    triggerMidGameJudgment,
    reset
  }
})
```

### 5.1 Axios实例配置

```typescript
// api/axios.ts
import axios from 'axios'
import type { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios'
import { ElMessage } from 'element-plus'

const axiosInstance: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
})

// 请求拦截器
axiosInstance.interceptors.request.use(
  (config) => {
    // 添加token
    const userStore = useUserStore()
    if (userStore.userId) {
      config.headers['X-User-ID'] = userStore.userId
    }
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// 响应拦截器
axiosInstance.interceptors.response.use(
  (response: AxiosResponse) => {
    return response.data
  },
  (error) => {
    // 统一错误处理
    if (error.response) {
      switch (error.response.status) {
        case 400:
          ElMessage.error('请求参数错误')
          break
        case 401:
          ElMessage.error('未授权，请重新登录')
          break
        case 403:
          ElMessage.error('拒绝访问')
          break
        case 404:
          ElMessage.error('请求资源不存在')
          break
        case 500:
          ElMessage.error('服务器错误')
          break
        default:
          ElMessage.error('网络错误')
      }
    } else if (error.request) {
      ElMessage.error('网络连接失败')
    } else {
      ElMessage.error('请求失败')
    }
    return Promise.reject(error)
  }
)

export default axiosInstance
```

### 6.2 开发建议

1. **先设计后编码**：在编码前先设计好组件结构和状态管理
2. **复用优先**：优先使用现有组件和Composables
3. **性能优化**：注意长列表渲染、频繁更新等性能问题
4. **错误处理**：所有异步操作都要有错误处理
5. **代码规范**：遵循统一的代码风格和命名规范

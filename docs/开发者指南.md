## 🧩 插件开发指南

### 创建自定义插件

#### 1. 继承 BasePlugin
```python
from alice.plugins.base_plugin import BasePlugin, PluginResult
from typing import Dict, Any

class MyCustomPlugin(BasePlugin):
    def __init__(self, config: Dict[str, Any] = None):
        super().__init__(config)
        self.priority = 75  # 设置插件优先级 (0-100)
    
    def initialize(self) -> bool:
        """初始化插件资源"""
        try:
            # 初始化必要的资源
            self.custom_data = load_custom_data()
            return True
        except Exception as e:
            self.logger.error(f"插件初始化失败: {e}")
            return False
    
    def process_input(self, text: str, context: Dict[str, Any]) -> PluginResult:
        """处理用户输入"""
        try:
            # 实现业务逻辑
            if self._should_process(text, context):
                response = self._generate_response(text, context)
                return PluginResult(
                    success=True,
                    response=response,
                    metadata={"processed_by": "my_custom_plugin"}
                )
            else:
                return PluginResult(success=False)
                
        except Exception as e:
            self.logger.error(f"插件处理失败: {e}")
            return PluginResult(success=False)
    
    def _should_process(self, text: str, context: Dict) -> bool:
        """判断是否应该处理此输入"""
        # 实现判断逻辑
        return "关键词" in text
    
    def _generate_response(self, text: str, context: Dict) -> str:
        """生成响应"""
        # 实现响应生成逻辑
        return f"关于'{text}'，我想说..."
```

## 📊 性能优化指南

### 1. 缓存策略
```python
from alice.cache import IntelligentCache

class OptimizedPlugin(BasePlugin):
    def __init__(self, config=None):
        super().__init__(config)
        self.cache = IntelligentCache(max_size=1000)
    
    def process_input(self, text: str, context: Dict) -> PluginResult:
        # 检查缓存
        cache_key = f"process_{hash(text)}"
        cached_result = self.cache.get(cache_key)
        
        if cached_result:
            return cached_result
        
        # 执行耗时操作
        result = self._expensive_operation(text)
        
        # 缓存结果
        self.cache.set(cache_key, result, ttl=3600)
        
        return result
```

### 2. 异步处理

import asyncio
from concurrent.futures import ThreadPoolExecutor

class AsyncPlugin(BasePlugin):
    def __init__(self, config=None):
        super().__init__(config)
        self.executor = ThreadPoolExecutor(max_workers=4)
    
    async def process_async(self, text: str, context: Dict) -> PluginResult:
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(
            self.executor,
            self._blocking_operation,
            text,
            context
        )

from queue import Queue

class ResourcePoolPlugin(BasePlugin):
    def __init__(self, config=None):
        super().__init__(config)
        self.resource_pool = Queue(maxsize=10)
        
        # 预分配资源
        for i in range(10):
            self.resource_pool.put(self._create_resource())
    
    def _get_resource(self):
        return self.resource_pool.get(timeout=5)
    
    def _return_resource(self, resource):
        self.resource_pool.put(resource)

    def __init__(self, config=None):

### 敏感信息处理
```python
import os
from cryptography.fernet import Fernet

class SecureStoragePlugin(BasePlugin):
    def __init__(self, config=None):
        super().__init__(config)
        self.encryption_key = os.getenv("ENCRYPTION_KEY")
        if self.encryption_key:
            self.cipher_suite = Fernet(self.encryption_key.encode())
    
    def _encrypt_sensitive_data(self, data: str) -> str:
        if self.cipher_suite:
            return self.cipher_suite.encrypt(data.encode()).decode()
        return data  # 降级处理
    
    def _decrypt_sensitive_data(self, encrypted_data: str) -> str:
        if self.cipher_suite:
            return self.cipher_suite.decrypt(encrypted_data.encode()).decode()
        return encrypted_data
```

### 日志记录
```python
import logging

class DebuggablePlugin(BasePlugin):
    def __init__(self, config=None):
        super().__init__(config)
        self.logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")
        
        # 设置日志级别
        debug_mode = self.config.get("debug", False)
        if debug_mode:
            self.logger.setLevel(logging.DEBUG)
    
    def process_input(self, text: str, context: Dict) -> PluginResult:
        self.logger.debug(f"处理输入: {text[:50]}...")
        
        try:
            result = self._process_logic(text, context)
            self.logger.info(f"处理成功，响应长度: {len(result.response)}")
            return result
        except Exception as e:
            self.logger.error(f"处理失败: {e}", exc_info=True)
            return PluginResult(success=False)
```

### 性能监控
```python
import time
from alice.utils.monitor import UnifiedMonitor

class MonitoredPlugin(BasePlugin):
    def __init__(self, config=None):
        super().__init__(config)
        self.monitor = UnifiedMonitor()
    
    def process_input(self, text: str, context: Dict) -> PluginResult:
        start_time = time.time()
        
        try:
            result = self._process_logic(text, context)
            
            # 记录性能数据
            duration = time.time() - start_time
            self.monitor.record_interaction(
                request_time=start_time,
                response_time=time.time(),
                success=True,
                metadata={
                    "plugin": self.name,
                    "input_length": len(text),
                    "duration_ms": duration * 1000
                }
            )
            
            return result
        except Exception as e:
            self.monitor.log_error("plugin_process", e, {
                "plugin": self.name,
                "input": text[:100]
            })
            raise
```

### 1.2 第一个 Lua 脚本（Hello World）

创建文件 `scripts/lua/hello.lua`：

```lua
-- 简单问候脚本
-- 匹配包含"你好"的消息并返回问候语

function match(context)
    local text = context.text or ""
    
    -- 检查是否包含问候关键词
    if string.find(text, "你好") then
        return {
            matched = true,
            priority = 90,
            confidence = 0.9
        }
    end
    
    return false
end

function generate_response(context)
    return "你好！很高兴见到你。"
end

-- 必须返回函数表
return { match = match, generate_response = generate_response }
```

```yaml
- intent: hello_greeting

  condition:
```

```python
    print(f"置信度：{match.confidence}")

    print(f"响应：{response.text}")

    print("未匹配")
```

## 第二章：脚本引擎架构

### 2.1 统一脚本引擎接口

所有脚本引擎继承自 `BaseScriptEngine`，提供统一接口：

```python
class BaseScriptEngine(ABC):
    @abstractmethod
    def load_script(self, config: ScriptConfig) -> bool:
        """加载脚本"""
        pass

    @abstractmethod
    def match(self, context: ScriptContext) -> Optional[ScriptMatchResult]:
        """匹配脚本"""
        pass

    @abstractmethod
    def generate_response(
        self, 
        script_id: str, 
        context: ScriptContext
    ) -> Optional[ScriptResponse]:
        """生成响应"""
        pass

    @abstractmethod
    def reload_script(self, script_id: str) -> bool:
        """热重载脚本"""
        pass
```

**优势**：
- Lua 和 YAML 脚本使用相同接口
- 易于扩展新引擎（如 JSON、Python 脚本）
- 统一的优先级调度

---

### 2.2 ScriptContext 数据结构

`ScriptContext` 是脚本执行的核心数据载体：

```python
@dataclass
class ScriptContext:
    # 输入数据
    text: str                              # 标准化文本
    tokens: List[str]                      # 分词结果
    entities: List[Tuple[str, str]]        # (类型，文本)
    syntax: Optional[Dict]                 # 句法结构
    pos_tags: List[Tuple[str, str]]        # (词，词性)
    dependencies: List[Dict]               # 依存关系
    triples: List[Tuple]                   # 语义三元组
    semantic_roles: List[Dict]             # 语义角色
    semantic_deps: List[Dict]              # 语义依存
    
    # 对话上下文
    turn_count: int                        # 对话轮数
    recent_turns: List[Dict]               # 最近对话
    time_context: Dict                     # 时间上下文
    user_profile: Dict                     # 用户画像
    
    # 脚本执行状态
    matched_scripts: List[str]             # 已匹配脚本
    variables: Dict                        # 脚本变量
```

**在 Lua 中访问**：

```lua
function match(context)
    local text = context.text
    local tokens = context.tokens          -- 表
    local entities = context.entities      -- 表
    
    -- 访问第一个实体
    if entities and #entities > 0 then
        local first_entity = entities[1]
        print(first_entity.type, first_entity.text)
    end
    
    -- 访问对话轮数
    local turns = context.turn_count
end
```

### 2.3 优先级调度机制

脚本匹配器 `ScriptMatcher` 统一管理所有引擎的脚本：

```
优先级范围    级别    用途
90-100       P0     非常重要 - 问候、告别、核心功能
70-89        P1     重要 - 实体挖掘、人物询问
40-69        P2     中等 - 叙事助推、通用回复
0-39         P3     低 - 万能回复、默认响应
```

**调度流程**：

```
1. 所有引擎并行/串行匹配
       ↓
2. 收集所有匹配结果
       ↓
3. 按 priority * 0.7 + confidence * 100 * 0.3 排序
       ↓
4. 选择最佳匹配
       ↓
5. 调用对应引擎生成响应
```

### 3.2 match() 函数详解

#### 返回值格式

**格式 1：布尔值**（简单匹配）

```lua
function match(context)
    if string.find(context.text, "你好") then
        return true  -- 使用默认优先级和置信度
    end
    return false
end
```

**格式 2：数字**（置信度）

```lua
function match(context)
    if string.find(context.text, "你好") then
        return 0.9  -- 置信度 0.9
    end
    return 0
end
```

**格式 3：详细信息表**（推荐）

```lua
function match(context)
    local score = calculate_match_score(context)
    
    if score > 0.5 then
        return {
            matched = true,
            priority = 85,           -- 动态优先级
            confidence = score,      -- 动态置信度
        }
    end
    return false
end
```

#### 匹配技巧

**关键词匹配**：

```lua
function match(context)
    local text = context.text or ""
    local keywords = {"你好", "您好", "嗨", "hello", "hi"}
    
    for _, kw in ipairs(keywords) do
        if string.find(text, kw, 1, true) then  -- true=精确匹配
            return true
        end
    end
    return false
end
```

**实体匹配**：

```lua
function match(context)
    if context.entities then
        for _, entity in ipairs(context.entities) do
            if entity.type == "PERSON" then
                return true  -- 匹配到人名
            end
            if entity.type == "LOCATION" then
                return true  -- 匹配到地点
            end
        end
    end
    return false
end
```

**对话轮数判断**：

```lua
function match(context)
    local turn_count = context.turn_count or 0
    
    if turn_count >= 10 then
        return true  -- 只匹配长对话
    end
    return false
end
```

```lua
function generate_response(context)
    local hour = tonumber(os.date("%H"))
```

```lua
function generate_response(context)

                person_name = entity.text
```

```lua
function generate_response(context)
```

```lua
function match(context)
```

#### 使用实体匹配

```lua
function match(context)
    -- 检查是否包含特定类型实体
    local function has_entity_type(entities, target_type)
        if not entities then return false end
        for _, e in ipairs(entities) do
            if e.type == target_type then
                return true
            end
        end
        return false
    end
    
    if has_entity_type(context.entities, "PERSON") then
        return { priority = 80, confidence = 0.8 }
    end
    
    return false
end
```

```lua
function generate_response(context)
```

### 3.7 高级示例

#### 情感分析脚本

```lua
-- scripts/lua/sentiment_analysis.lua
-- 基于关键词的情感分析

local positive_words = {
    "开心", "高兴", "快乐", "幸福", "美好",
    "喜欢", "爱", "棒", "好", "赞",
}

local negative_words = {
    "难过", "伤心", "痛苦", "糟糕", "差",
    "讨厌", "恨", "烦", "累", "困",
}

function match(context)
    local text = context.text or ""
    local score = 0
    
    -- 计算情感得分
    for _, word in ipairs(positive_words) do
        if string.find(text, word, 1, true) then
            score = score + 1
        end
    end
    
    for _, word in ipairs(negative_words) do
        if string.find(text, word, 1, true) then
            score = score - 1
        end
    end
    
    if score ~= 0 then
        -- 存储情感得分到上下文变量
        context.variables.sentiment_score = score
        return {
            matched = true,
            priority = 70,
            confidence = math.min(math.abs(score) / 3, 1.0),
        }
    end
    
    return false
end

function generate_response(context)
    local score = context.variables.sentiment_score or 0
    
    if score > 0 then
        return "听起来你心情不错！有什么好事吗？"
    elseif score < 0 then
        return "感觉你心情不太好，想聊聊吗？"
    else
        return "嗯，我明白了。"
    end
end

return { match = match, generate_response = generate_response }
```

#### 对话流程控制

```lua
-- scripts/lua/conversation_flow.lua
-- 管理对话流程和状态

local MAX_TURNS = 20
local state = {
    topic = nil,
    question_count = 0,
}

function match(context)
    local turn_count = context.turn_count or 0
    
    -- 检查是否需要结束对话
    if turn_count >= MAX_TURNS then
        return {
            matched = true,
            priority = 95,
            confidence = 1.0,
        }
    end
    
    -- 检查是否应该追问
    if state.question_count < 3 then
        return {
            matched = true,
            priority = 60,
            confidence = 0.7,
        }
    end
    
    return false
end

function generate_response(context)
    local turn_count = context.turn_count or 0
    
    if turn_count >= MAX_TURNS then
        return {
            text = "我们已经聊了很久了，下次再聊吧！",
            end_action = "farewell",
            end_reason = "bot_max_turns",
        }
    end
    
    -- 追问逻辑
    state.question_count = state.question_count + 1
    
    local followups = {
        "然后呢？",
        "能再多说一些吗？",
        "这很有意思，后来怎么样了？",
        "为什么会这样呢？",
    }
    
    return followups[(state.question_count - 1) % #followups + 1]
end

return { match = match, generate_response = generate_response }
```

### 4.2 意图定义（intent）

`intent` 是意图的唯一标识：

```yaml
# 问候意图
- intent: greeting
```

**命名规范**：
- 使用小写字母和下划线
- 语义清晰，如 `greeting` 而非 `greet`
- 避免中文，使用英文标识

### 7.1 人机辨识博弈脚本（图灵测试场景）

#### 场景说明

在图灵测试中，Bot 需要：
1. 模拟人类行为模式
2. 避免过于完美的回答
3. 适当表现"不确定"

#### Lua 实现

```lua
-- scripts/lua/turing_bot.lua
-- 图灵测试 Bot 脚本

local human_like_responses = {
    uncertain = {
        "嗯...让我想想",
        "这个嘛，不太确定",
        "好像是这样？",
        "我记得应该是吧",
    },
    thinking = {
        "好问题，我需要思考一下",
        "这有点复杂",
        "让我回忆一下",
    },
}

function match(context)
    -- 检测是否需要人类式回应
    local text = context.text or ""
    
    -- 复杂问题触发人类式回应
    local complex_keywords = {
        "为什么", "怎么", "如何", "什么", "谁",
    }
    
    for _, kw in ipairs(complex_keywords) do
        if string.find(text, kw, 1, true) then
            return {
                matched = true,
                priority = 75,
                confidence = 0.8,
            }
        end
    end
    
    return false
end

function generate_response(context)
    -- 随机选择人类式回应
    local response_type = "uncertain"
    local responses = human_like_responses[response_type]
    
    -- 添加一些个人化元素
    local personal_touches = {
        "我觉得",
        "我个人认为",
        "以我的经验",
        "据我所知",
    }
    
    local response = responses[math.random(#responses)]
    local touch = personal_touches[math.random(#personal_touches)]
    
    return touch .. "，" .. response
end

return { match = match, generate_response = generate_response }
```

### 7.2 多轮对话状态管理

#### 状态机设计

```lua
-- scripts/lua/state_machine.lua
-- 多轮对话状态管理

local StateMachine = {
    current_state = "idle",
    states = {
        "idle",
        "asking_name",
        "asking_age",
        "asking_location",
        "complete",
    },
}

local transitions = {
    idle = { start_chat = "asking_name" },
    asking_name = { got_name = "asking_age" },
    asking_age = { got_age = "asking_location" },
    asking_location = { got_location = "complete" },
    complete = { reset = "idle" },
}

function match(context)
    -- 检查当前状态
    local vars = context.variables
    local state = vars.chat_state or "idle"
    
    -- 根据状态匹配
    if state == "asking_name" then
        -- 检查是否包含人名
        if context.entities then
            for _, e in ipairs(context.entities) do
                if e.type == "PERSON" then
                    return { matched = true, priority = 90 }
                end
            end
        end
    end
    
    return false
end

function generate_response(context)
    local vars = context.variables
    local state = vars.chat_state or "idle"
    
    if state == "idle" then
        vars.chat_state = "asking_name"
        return "你好！请问怎么称呼？"
        
    elseif state == "asking_name" then
        -- 获取人名
        local name = nil
        if context.entities then
            for _, e in ipairs(context.entities) do
                if e.type == "PERSON" then
                    name = e.text
                    break
                end
            end
        end
        
        if name then
            vars.user_name = name
            vars.chat_state = "asking_age"
            return string.format("%s，很高兴认识你！请问你今年多大了？", name)
        end
        
    elseif state == "asking_age" then
        vars.chat_state = "asking_location"
        return "原来是这样，那你住在哪个城市呢？"
        
    elseif state == "asking_location" then
        vars.chat_state = "complete"
        return "好的，我记住了！很高兴和你聊天。"
    end
    
    return "嗯，我明白了。"
end

return { match = match, generate_response = generate_response }
```

```lua
-- 用户记忆（持久化需要外部存储）
local user_memories = {}

function match(context)
    local text = context.text or ""
    
    -- 检查是否包含问候 + 人名
    if string.find(text, "你好") and context.entities then
        for _, e in ipairs(context.entities) do

                return {
                    matched = true,
                    priority = 95,

    if context.entities then
        for _, e in ipairs(context.entities) do
            if e.type == "PERSON" then
                name = e.text
                break
            end
        end
    end
    
    -- 从记忆中获取上次对话信息
    local last_seen = user_memories[name]
    
    if name then
        if last_seen then
            -- 老用户
            local responses = {
                string.format("%s，又见面了！上次我们聊到%s，后来怎么样了？",

        else
            -- 新用户
            user_memories[name] = {
                first_seen = os.date("%Y-%m-%d"),
                topic = nil,
            }
            return string.format("%s，你好！很高兴认识你！", name)
        end
    end
    
    return "你好！请问怎么称呼？"
end

return { match = match, generate_response = generate_response }
```

### D. 常见错误排查

#### 1. Lua 引擎不可用

**错误**：`ImportError: lupa 库未安装`

**解决**：
```bash
uv add lupa
```

#### 2. YAML 脚本加载失败

**错误**：`YAML 解析错误`

**解决**：
- 检查 YAML 缩进（使用空格，不用 Tab）
- 使用 YAML 验证工具检查语法
- 确保列表项使用 `-` 开头

#### 3. 脚本匹配失败

**排查步骤**：
```python
# 1. 检查脚本是否加载
scripts = engine.list_scripts()
print(scripts)

# 2. 检查优先级设置
stats = engine.get_stats()
print(stats['engines'])

# 3. 启用调试日志
# config.yaml: log_level: DEBUG
```

#### 4. NLP 分析结果为空

**原因**：LTP 模型未加载

**解决**：
```yaml
# config.alice.yaml
alice:
  enable_ltp: true
  # 确保模型路径正确
  ltp_model_path: "path/to/ltp/models"
```

#### 5. 沙箱超时

**错误**：`TimeoutError: 脚本执行超时`

**解决**：
```yaml
# metadata.yaml
scripts:
  - name: complex_script
    max_execution_time: 5.0  # 增加超时时间
```

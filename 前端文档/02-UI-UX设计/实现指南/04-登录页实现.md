# 登录页实现

## 文档说明

本文档详细说明图灵测试实验平台登录页面的实现，包括邀请码登录、昵称设置、表单验证等功能。

---

## 一、页面概述

### 1.1 功能说明

登录页是用户进入平台的第一步，主要功能包括：

1. **邀请码登录**
   - 6位字母数字组合
   - 大小写不敏感
   - 前后去除空格
   - 后端验证有效性

2. **昵称设置**
   - 登录成功后弹出
   - 长度2-20个字符
   - 支持中文、英文、数字
   - 默认随机昵称

### 3.2 LoginForm.vue - 登录表单

```vue
<template>
  <form @submit.prevent="handleSubmit" class="login-form">
    <div class="form-group">
      <label for="inviteCode">邀请码</label>
      <input
        id="inviteCode"
        v-model="formData.inviteCode"
        type="text"
        maxlength="6"
        placeholder="输入6位邀请码"
        :disabled="loading"
        class="input-field"
        @input="handleInput"
      />
      <p v-if="errors.inviteCode" class="error-message">
        {{ errors.inviteCode }}
      </p>
    </div>
    
    <button
      type="submit"
      :disabled="loading"
      class="btn btn-primary btn-large"
    >
      <span v-if="loading" class="spinner"></span>
      <span v-else>登录</span>
    </button>
  </form>
</template>

<script setup lang="ts">
import { ref, reactive } from 'vue'
import { login } from '@/api/auth'
import { ElMessage } from 'element-plus'

const emit = defineEmits<{
  (e: 'login-success', data: { userId: string; inviteCode: string }): void
}>()

const formData = reactive({
  inviteCode: ''
})

const errors = reactive({
  inviteCode: ''
})

const loading = ref(false)

// 处理输入
function handleInput(event: Event) {
  const target = event.target as HTMLInputElement
  // 只允许字母和数字
  target.value = target.value.toUpperCase().replace(/[^A-Z0-9]/g, '')
  // 限制长度为6
  if (target.value.length > 6) {
    target.value = target.value.slice(0, 6)
  }
  formData.inviteCode = target.value
  
  // 清除错误提示
  errors.inviteCode = ''
}

// 表单验证
function validateForm(): boolean {
  errors.inviteCode = ''
  
  if (!formData.inviteCode) {
    errors.inviteCode = '请输入邀请码'
    return false
  }
  
  if (formData.inviteCode.length !== 6) {
    errors.inviteCode = '邀请码必须是6位'
    return false
  }
  
  return true
}

// 提交登录
async function handleSubmit() {
  // 1. 前端验证
  if (!validateForm()) {
    return
  }
  
  // 2. 开始加载
  loading.value = true
  
  try {
    // 3. 调用API
    const response = await login({
      inviteCode: formData.inviteCode
    })
    
    // 4. 登录成功
    ElMessage.success('登录成功')
    
    emit('login-success', {
      userId: response.user_id,
      inviteCode: formData.inviteCode
    })
    
  } catch (error: any) {
    // 5. 处理错误
    console.error('登录失败:', error)
    
    if (error.response?.data?.error === 'invalid_code') {
      errors.inviteCode = '邀请码无效，请检查输入'
    } else if (error.response?.data?.error === 'invite_code_used') {
      errors.inviteCode = '该邀请码已被使用'
    } else {
      ElMessage.error('登录失败，请稍后重试')
    }
    
  } finally {
    // 6. 结束加载
    loading.value = false
  }
}
</script>
```

```vue
<script setup lang="ts">

    // 4. 设置成功

  } catch (error) {

    console.error('设置昵称失败:', error)
    ElMessage.error('设置昵称失败，请稍后重试')
    
  } finally {
    // 6. 结束加载
    loading.value = false
  }
}

// 点击遮罩层关闭
function handleOverlayClick() {
  if (!loading.value) {
    emit('update:visible', false)
  }
}

// 监听visible变化，重置表单
watch(() => props.visible, (newVal) => {
  if (!newVal) {
    formData.nickname = ''
    errors.nickname = ''
  }
})
</script>
```

## 五、表单验证规则

### 5.1 邀请码验证

```typescript
// utils/validators.ts
export function validateInviteCode(code: string): { valid: boolean; message: string } {
  // 1. 非空检查
  if (!code) {
    return { valid: false, message: '请输入邀请码' }
  }
  
  // 2. 长度检查
  if (code.length !== 6) {
    return { valid: false, message: '邀请码必须是6位' }
  }
  
  // 3. 格式检查
  const regex = /^[A-Z0-9]{6}$/
  if (!regex.test(code)) {
    return { valid: false, message: '邀请码格式不正确' }
  }
  
  return { valid: true, message: '' }
}
```

### 5.2 昵称验证

```typescript
export function validateNickname(nickname: string): { valid: boolean; message: string } {
  // 昵称是可选的
  if (!nickname) {
    return { valid: true, message: '' }
  }
  
  // 1. 长度检查
  if (nickname.length < 2) {
    return { valid: false, message: '昵称长度不能少于2个字符' }
  }
  
  if (nickname.length > 20) {
    return { valid: false, message: '昵称长度不能超过20个字符' }
  }
  
  // 2. 格式检查（中文、英文、数字、#和数字）
  const regex = /^[\u4e00-\u9fa5a-zA-Z0-9#]+$/
  if (!regex.test(nickname)) {
    return { valid: false, message: '昵称只能包含中文、英文、数字和#' }
  }
  
  return { valid: true, message: '' }
}
```

# 场中判断机制实现

## 文档说明

本文档详细说明场中判断机制的实现，这是图灵测试实验平台的高风险高收益功能。

---

## 一、功能概述

### 1.1 场中判断定义

场中判断是指用户可以在对话过程中的任何时间点主动进行判断，无需等待对话结束。

**核心特点**：
1. **主动触发**：用户随时可以点击"场中判断"按钮
2. **双倍奖励**：判断正确时获得双倍奖励（基础分 × 2.0）
3. **1.5倍惩罚**：判断错误时受到1.5倍惩罚（基础分 × 1.5）
4. **立即结束**：判断后对话立即结束，无需填写问卷

### 1.2 策略意义

```
场中判断 = 高风险 × 高收益 × 快速结束

适用场景：
- 确信度高时：快速锁定收益
- 避免轮数惩罚：减少因时间过长而损失积分
- 战术性选择：在最佳时机进行判断
```

## 二、Composable实现

### 2.1 useMidGameJudgment.ts

```typescript
// composables/useMidGameJudgment.ts
import { computed } from 'vue'
import { useGameStore } from '@/stores/game'
import { useScorePrediction } from '@/composables/useScorePrediction'

export function useMidGameJudgment() {
  const gameStore = useGameStore()
  const { metaMultiplier, turnPenalty } = useScorePrediction()

  // 场中判断倍数
  const MID_GAME_CORRECT_MULTIPLIER = 2.0  // 双倍奖励
  const MID_GAME_WRONG_MULTIPLIER = 1.5     // 1.5倍惩罚

  // 基础分
  const BASE_REWARD = 10      // 判断正确
  const BASE_PENALTY = -15    // 判断错误（AI）
  const ENTRY_FEE = 2

  // 计算场中判断的积分预测
  const midGamePrediction = computed(() => {
    const metaMult = metaMultiplier.value
    const penalty = turnPenalty.value

    const correctScore = 
      (BASE_REWARD * MID_GAME_CORRECT_MULTIPLIER * metaMult) - ENTRY_FEE - penalty
    const wrongScore = 
      (BASE_PENALTY * MID_GAME_WRONG_MULTIPLIER * metaMult) - ENTRY_FEE - penalty

    return {
      correct: correctScore.toFixed(1),
      wrong: wrongScore.toFixed(1),
      multiplier: `×${metaMult.toFixed(1)} (场中双倍)`,
      riskLevel: wrongScore < -100 ? '高风险' : '中等风险'
    }
  })

  // 判断是否可以场中判断
  const canMidGameJudge = computed(() => {
    // 至少需要1轮对话
    return gameStore.turn >= 1
  })

  // 获取风险等级颜色
  const getRiskColor = (level: string) => {
    switch(level) {
      case '高风险':
        return '#f56c6c'
      case '中等风险':
        return '#e6a23c'
      case '低风险':
        return '#67c23a'
      default:
        return '#409eff'
    }
  }

  return {
    midGamePrediction,
    canMidGameJudge,
    getRiskColor
  }
}
```

import { computed } from 'vue'
import { useRouter } from 'vue-router'

import { useMidGameJudgment } from '@/composables/useMidGameJudgment'

  visible: boolean

  (e: 'update:visible', value: boolean): void
}>()

const sessionStore = useSessionStore()

  midGamePrediction,
  getRiskColor
} = useMidGameJudgment()

const loading = ref(false)

const turn = computed(() => gameStore.turn)
const metaCount = computed(() => gameStore.metaConversationCount)
const metaMultiplier = computed(() => 1 + metaCount.value * 0.2)

// 执行场中判断
async function handleJudgment(guess: 'human' | 'ai') {
  loading.value = true
  
  try {
    // 1. 调用API
    const result = await makeMidGameJudgment({
      sessionId: gameStore.sessionId,
      userGuess: guess
    })
    
    // 2. 更新用户积分
    const userStore = useUserStore()
    userStore.updateScore(result.score)
    userStore.incrementSessionCount()
    
    // 3. 保存结果到Session Store
    sessionStore.surveyResult = {
      ...result,
      isMidGame: true,
      userGuess: guess
    }
    
    // 4. 关闭弹窗
    emit('update:visible', false)
    
    // 5. 跳转到结果页
    router.push('/result')
    
    ElMessage.success('判断完成')
    
  } catch (error) {
    console.error('场中判断失败:', error)
    ElMessage.error('判断失败，请稍后重试')
  } finally {
    loading.value = false
  }
}

// 取消场中判断
function handleCancel() {
  emit('update:visible', false)
}

// 点击遮罩层关闭
function handleOverlayClick() {
  if (!loading.value) {
    handleCancel()
  }
}
</script>

## 五、API接口

### 5.1 场中判断接口

```typescript
// api/judgment.ts
export interface MidGameJudgmentRequest {
  sessionId: string
  userGuess: 'human' | 'ai'
}

export interface MidGameJudgmentResponse {
  success: boolean
  is_correct: boolean
  opponent_type: 'human' | 'ai' | 'honeypot'
  score: number
  score_breakdown: {
    base_score: number
    mid_game_multiplier: number
    meta_multiplier: number
    entry_fee: number
    turn_penalty: number
    final_score: number
  }
  current_total_score: number
}

export function makeMidGameJudgment(data: MidGameJudgmentRequest) {
  return request.post<MidGameJudgmentResponse>('/judgment/mid-game', data)
}
```

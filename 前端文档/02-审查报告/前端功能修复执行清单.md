// 结束对话
async function handleEndChat() {
  // 检查最低轮数
  if (gameStore.gameState.turn < MIN_CHAT_TURNS) {
    ElMessage.warning(`请多聊几句再结束哦（至少${MIN_CHAT_TURNS}轮，当前${gameStore.gameState.turn}轮）`)
    return
  }
  
  // 确认对话框
  const confirmed = await ElMessageBox.confirm(
    '确定要结束这次对话吗？',
    '结束对话',
    {
      confirmButtonText: '确定',
      cancelButtonText: '取消',
      type: 'warning'
    }
  )
  
  if (confirmed) {
    try {
      await endSession(gameStore.sessionId)
      router.push('/survey')
    } catch (error) {
      ElMessage.error('结束对话失败，请重试')
    }
  }
}
```

// 添加验证函数
function validateNickname(nick: string): boolean {
  // 长度检查
  if (nick.length < 2 || nick.length > 20) {
    errorMessage.value = '昵称长度应为 2-20 个字符'
    return false
  }
  
  // 格式检查（仅允许中文、英文、数字、#）
  const nicknameRegex = /^[\u4e00-\u9fa5a-zA-Z0-9#]+$/
  if (!nicknameRegex.test(nick)) {
    errorMessage.value = '昵称仅支持中文、英文、数字和#符号'
    return false
  }
  
  return true
}

// 在 registerUser 中调用
async function registerUser() {
  if (!nickname.value.trim()) {
    errorMessage.value = '请输入昵称'
    return
  }
  
  // 新增：格式验证
  if (!validateNickname(nickname.value.trim())) {
    return
  }
  
  loading.value = true
  // ...
}

// utils/formatters.ts
export function stringToColor(str: string): string {
  let hash = 0
  for (let i = 0; i < str.length; i++) {
    hash = str.charCodeAt(i) + ((hash << 5) - hash)
  }
  
  // 生成 HSL 颜色
  const hue = hash % 360
  return `hsl(${hue}, 70%, 50%)`
}

export function getInitials(name: string): string {
  return name.charAt(0).toUpperCase()
}
```
